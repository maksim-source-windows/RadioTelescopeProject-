===============================================
           УРОВЕНЬ 1: РУЧНОЕ С МОТОРОМ
===============================================

Самый простой вариант — добавить мотор для одной оси.

КОМПОНЕНТЫ:
- Сервопривод MG996R или DS3218 (момент 20+ кг/см)
- Крепёж для серво к параболе
- Отдельный источник питания 5-6 В, не менее 3 А на серво (MG996R в упоре до ~2.5 А)
- Провода

РАСЧЁТ МОМЕНТА:
Вес параболы ~2 кг
Плечо (от центра до крепления) ~30 см

Момент = сила × плечо
M = 2 кг × 9.8 м/с² × 0.3 м = 5.9 Н·м

MG996R: 20 кг/см = 1.96 Н·м - НЕДОСТАТОЧНО для 2 кг параболы!
Требуется либо противовес, либо серво с моментом > 6 Н·м.

СХЕМА ПОДКЛЮЧЕНИЯ:
Arduino:
  D9  → сигнал серво
  Питание серво: ОТ ВНЕШНЕГО БП 5-6 В (не от пина 5V Arduino: просадка перезагрузит плату)
  GND → общая земля БП серво и Arduino

КОД (уровень 1):
-----------------------------------
#include <Servo.h>

Servo myservo;
int pos = 90; // начальный угол

void setup() {
  myservo.attach(9);
  myservo.write(90);
}

void loop() {
  // медленное сканирование
  for (pos = 0; pos <= 180; pos += 1) {
    myservo.write(pos);
    delay(50);
  }
  for (pos = 180; pos >= 0; pos -= 1) {
    myservo.write(pos);
    delay(50);
  }
}
-----------------------------------

СТОИМОСТЬ: ~300-500 ₽


===============================================
         УРОВЕНЬ 2: ДВЕ ОСИ С ДЖОЙСТИКОМ
===============================================

Управление влево-вправо и вверх-вниз джойстиком.

КОМПОНЕНТЫ:
- 2 сервопривода MG996R
- 1 джойстик XY
- Arduino Nano
- Отдельный блок питания 6 В, не менее 5 А (две MG996R)
  (земля БП соединена с GND Arduino)

ПОДКЛЮЧЕНИЕ:
Arduino:
  A0 → джойстик X (горизонт)
  A1 → джойстик Y (вертикаль)
  D9 → серво горизонт
  D10 → серво вертикаль

КОД (уровень 2):
-----------------------------------
#include <Servo.h>

Servo servoH, servoV;
int joyX, joyY;
float angleH = 90.0, angleV = 45.0;  // float для точного накопления

void setup() {
  servoH.attach(9);
  servoV.attach(10);
  servoH.write((int)angleH);
  servoV.write((int)angleV);
}

void loop() {
  joyX = analogRead(A0);
  joyY = analogRead(A1);

  // мёртвая зона
  if (abs(joyX - 512) > 10) {
    float delta = (joyX - 512) / 200.0;
    angleH += delta;
    angleH = constrain(angleH, 0, 180);
    servoH.write((int)(angleH + 0.5));  // округление
  }

  if (abs(joyY - 512) > 10) {
    float delta = (joyY - 512) / 200.0;
    angleV -= delta;
    angleV = constrain(angleV, 10, 80);
    servoV.write((int)(angleV + 0.5));  // округление
  }

  delay(20);
}
-----------------------------------

СТОИМОСТЬ: ~600-800 ₽


===============================================
         УРОВЕНЬ 3: АВТОНАВЕДЕНИЕ С КООРДИНАТАМИ
===============================================

Наведение по небесным координатам (азимут, высота).

КОМПОНЕНТЫ:
- 2 сервопривода
- 2 энкодера (дают обратную связь по углу)
- GPS модуль (определение местоположения)
- Arduino Mega (или Nano + Serial)
- RTC модуль (время)

ЭНКОДЕР:
- Магнитный энкодер AS5600 (I2C, 12 бит, ~0.09°) или KY-040 с редуктором
- KY-040 без редуктора: ~20 импульсов на оборот, то есть шаг 18° (80 отсчётов
  квадратурного декодирования = 4.5°): для точности ~1° не подходит

ПОДКЛЮЧЕНИЕ:
  D2  → энкодер 1 (горизонт)
  D3  → энкодер 2 (вертикаль)
  D9  → серво горизонт
  D10 → серво вертикаль
  D4  → GPS TX (RX программного порта)
  D5  → GPS RX (TX программного порта)

АЛГОРИТМ:
1. Получить GPS координаты и время
2. Вычислить азимут и высоту объекта (Альт-Азимут)
3. Перевести в углы серво
4. Повернуть
5. Проверить энкодеры — довернуть если нужно

КОД (уровень 3) - ЭСКИЗ, НЕ КОМПИЛИРУЕТСЯ КАК ЕСТЬ:
calculateAltAz() нужно реализовать (например, на ПК через astropy, а на
Arduino слать готовые "SET az alt"). Азимут 0-360° на обычный серво (180°)
не отображается: используйте сектор 0-180° или шаговик с редуктором.
-----------------------------------
#include <Servo.h>
#include <TinyGPS++.h>
#include <SoftwareSerial.h>

Servo servoH, servoV;
TinyGPSPlus gps;

SoftwareSerial ss(4, 5); // RX, TX для GPS (D3 занят энкодером; на Mega лучше Serial1)

// Небесные координаты (например, центр Галактики)
float targetRA = 266.4;     // прямое восхождение, град (17h45m)
float targetDec = -29.0;    // склонение, град

void setup() {
  Serial.begin(9600);
  ss.begin(9600);
  servoH.attach(9);
  servoV.attach(10);
}

void loop() {
  while (ss.available())
    gps.encode(ss.read());

  if (gps.location.isValid() && gps.time.isValid()) {  // fix есть (работает и в южном полушарии)
    // Вычислить азимут и высоту
    float alt, az;
    calculateAltAz(gps.location.lat(), gps.location.lng(),
                   gps.date.value(), gps.time.value(),
                   targetRA, targetDec, &alt, &az);

    // Перевести в углы серво
    int servoH_angle = constrain((int)az, 0, 180)  /* сектор 0-180°, не map() 0-360 */;
    int servoV_angle = constrain((int)alt, 10, 80);  // при калибровке 0° серво = горизонт

    servoH.write(servoH_angle);
    servoV.write(servoV_angle);

    Serial.print("AZ: "); Serial.print(az);
    Serial.print("  ALT: "); Serial.println(alt);
  }

  delay(100);
}

===============================================
          УРОВЕНЬ 4: ПОЛНОЕ АВТОСКАНИРОВАНИЕ
===============================================

Автоматическое сканирование неба и построение карты.

КОМПОНЕНТЫ:
- Всё из уровня 3
- Подключение к RTL-SDR (Python)
- SD карта для записи данных

АЛГОРИТМ СКАНИРОВАНИЯ:
for (az = 0; az <= 180; az += step) {      // серво: сектор 0-180°
  for (alt = 10; alt <= 80; alt += step) {
    навести(az, alt);
    delay(интеграция);        // 5-30 сек для проверки, 300 сек для науки
    signal = read_SDR();      // получить уровень
    save(az, alt, signal);    // записать
  }
}

ВРЕМЯ СКАНИРОВАНИЯ:
Шаг 10° × 10°:
- 19 азимутов × 8 высот = 152 точки
- По 10 сек на точку и ~3 сек на ход = ~33 минуты
- По 300 сек на точку: ~13 часов (небо повернётся на ~190°, см. auto_scanner.py)

СХЕМА СВЯЗИ ARDUINO ↔ ПК:
Arduino → USB → Python скрипт
Команды: "SET <az> <alt>" -> ответ "POS <az> <alt>" (как в software/arduino_telescope.ino).
Сигнал читает ПК с RTL-SDR (software/auto_scanner.py), не Arduino.

ПРИМЕР PYTHON (использует software/auto_scanner.py):
-----------------------------------
import numpy as np
from auto_scanner import RadioTelescope, SETTINGS, plot_skymap

azimuths = list(range(0, 181, 10))
elevations = list(range(10, 81, 10))

tel = RadioTelescope(SETTINGS['serial_port'])
data = np.full((len(elevations), len(azimuths)), np.nan)
try:
    for i, alt in enumerate(elevations):
        for j, az in enumerate(azimuths):
            # SET <az> <alt> -> POS, затем интеграция и анализ линии
            ratio_db, ok, t, spec = tel.measure_point(az, alt, 10)
            data[i, j] = ratio_db if ok else np.nan
finally:
    tel.close()

plot_skymap(data, 'skymap.png', azimuths, elevations)
-----------------------------------
